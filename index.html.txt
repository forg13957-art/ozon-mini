<script>
  const products = [
    { id: 1, name: "Смартфон", price: 19990, img: "https://via.placeholder.com/200" },
    { id: 2, name: "Ноутбук", price: 59990, img: "https://via.placeholder.com/200" },
    { id: 3, name: "Наушники", price: 2990, img: "https://via.placeholder.com/200" },
    { id: 4, name: "Часы", price: 4990, img: "https://via.placeholder.com/200" },
  ];

  let cart = 0;

  const container = document.getElementById("products");
  const cartCount = document.getElementById("cartCount");
  const searchInput = document.getElementById("search");

  function render(items) {
    container.innerHTML = "";
    items.forEach(p => {
      container.innerHTML += `
        <div class="card">
          <img src="${p.img}" />
          <div class="title">${p.name}</div>
          <div class="price">${p.price} ₽</div>
          <button onclick="addToCart()">В корзину</button>
        </div>
      `;
    });
  }

  function addToCart() {
    cart++;
    cartCount.innerText = cart;
  }

  searchInput.addEventListener("input", () => {
    const value = searchInput.value.toLowerCase();
    const filtered = products.filter(p =>
      p.name.toLowerCase().includes(value)
    );
    render(filtered);
  });

  render(products);
</script>
